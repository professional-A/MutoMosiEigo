# 武藤模試v2 M1（エンジン基盤＋新スキーマ＋プレイヤー）実装計画

> **実行エージェントへ:** REQUIRED SUB-SKILL: この計画は superpowers:executing-plans でタスク順に実行すること。ステップは checkbox（`- [ ]`）で進捗管理。1タスク1コミット。作業リポジトリは **現行の MutoMosiEigo**（`C:\Users\kabu6\MutoMosiEigo`）。

**Goal:** 現行武藤模試（Express + Render）の中に、新しい問題パックスキーマ（5形式＋例題）で動くv2エンジン（Vite + React サブアプリ）を `/v2` として組み込み、「解答→即時フィードバック→結果画面」まで通す。解答は既存ポイントAPIと新設の学習記録テーブルに送信する。

**Architecture:** リポジトリ内 `engine/` に独立したVite + React 18 + TSサブプロジェクトを作り、ビルド成果物（`engine/dist`）をExpressが `/v2` で配信する。既存ページ・既存テストは一切変更しない。認証は既存の `localStorage.muto_session` トークンを読むだけ（同一オリジンなので共有される）。将来 `engine/` フォルダごと切り出して単体配布・販売できる構造を保つ。

**Tech Stack:** Vite / React 18 / TypeScript / Tailwind CSS / KaTeX（npm同梱）/ Vitest / react-router-dom（HashRouter）。サーバー側は既存の Express + pg のまま。

## 背景（Context）

- 当初は完全新規のスタンドアロンアプリ（仕様書v2.1の想定）だったが、**方針転換**：現行武藤模試には楽しんで使っているユーザー（クラスメート）が多数いるため、**現行アプリ（Render配信・ポイント/バトル/レース込み）にv2エンジンを組み込み、テストの作り方を刷新する**プロジェクトにする。プロジェクト終了後にエンジン部分だけ切り出して配布・販売する
- テスト作成方法は**新パックスキーマに全面刷新**。現行の `data.json`（single/multi/sort/input）は将来廃止し、**英語模試の良い部分**（セクション/カテゴリ構成・入力正規化 `norm()`・語彙入力形式・並べ替え形式）を新スキーマに取り込む
- **収益化等に向けた学習記録は惜しみなく収集**する（Supabaseに解答ログを蓄積。M1から開始）
- ダッシュボード・ストリークも後続マイルストーンで積極的に実装する
- 全体仕様の学習科学的な根拠・機能定義は `mutomoshi_engine_仕様書_v2.1.md` と `武藤模試v2 開発のための2大テーマ・ディープリサーチ.md`（リポジトリルート）を参照。ただし「完全クライアントサイド・localStorage保存」等のスタンドアロン前提は本方針転換により**サーバー保存優先**に読み替える

```ts
// 問題パック v2 スキーマ（仕様書§2 ＋ 英語模試の形式を統合）
export type QuestionType = 'mc' | 'fill' | 'qa' | 'steps' | 'sort'

export interface WorkedExample {
  problem: string
  solution: string
}

interface QuestionBase {
  id: string // パック内一意。学習記録・重複排除キー（questionKey相当）
  category: string
  difficulty?: number // 1〜3想定
  source?: string // 例: "ノートp.3"
  explanation?: string
  workedExample?: WorkedExample // 学習モードで問題の前に例題として表示
}

// 4択（英語模試のWORDCHOICE/TF/QA形式もこれで表現。TFは2択）
// answerは0始まりindex。表示時シャッフル＆再マップ
export interface McQuestion extends QuestionBase {
  type: 'mc'
  question: string
  choices: string[]
  answer: number
}

// 穴埋め・語彙入力（英語模試のVOCAB形式）
// questionに {{blank}} があれば穴埋め表示、無ければ「定義→入力」形式
// TODO(仕様確認): {{blank}} 複数個は未対応（M1は0〜1個）
export interface FillQuestion extends QuestionBase {
  type: 'fill'
  question: string
  answer: string
}

// 記述・自己採点。answerは模範解答
export interface QaQuestion extends QuestionBase {
  type: 'qa'
  question: string
  answer: string
}

export interface QuizStep {
  text: string // {{blank}} を1つ含む
  answer: string
  why?: string // 自己説明を促す一言。解答後に表示
}

// 大問ステップ穴埋め（工学系の導出・計算問題向け）
export interface StepsQuestion extends QuestionBase {
  type: 'steps'
  title: string
  intro?: string
  steps: QuizStep[]
}

// 並べ替え（英語模試のWRITING形式）
export interface SortQuestion extends QuestionBase {
  type: 'sort'
  question: string // 指示文や和訳（英語模試のjp相当）
  pre?: string // 並べ替え部分の前に固定表示する語句
  post?: string // 後に固定表示する語句
  bank: string[] // 語群（表示時シャッフル）
  answer: string[] // 正しい並び順（並べ替え部分のみ）
}

export type Question = McQuestion | FillQuestion | QaQuestion | StepsQuestion | SortQuestion

export interface QuizPack {
  version: 2
  packId: string // 全体で一意。ポイント重複排除のquizKey・学習記録キーに使う
  subject: string // 表示名。既存の window.QUIZ_SUBJECT / races.subject と同じ値体系
  title: string
  examDate?: string // YYYY-MM-DD。スケジューラ（M2）が使用
  pointsPerQ?: number // 1問正解あたりのポイント。省略時100
  categories: string[]
  questions: Question[]
}
```

```ts
export function normalizeAnswer(s: string): string {

    .trim()
```

```ts
// engine/src/lib/shuffle.ts
// Fisher–Yates。rngはテスト用に注入可能
export function shuffleArray<T>(items: T[], rng: () => number = Math.random): T[] {
  const arr = [...items]
  for (let i = arr.length - 1; i > 0; i--) {
    const j = Math.floor(rng() * (i + 1))
    ;[arr[i], arr[j]] = [arr[j], arr[i]]
  }
  return arr
}

// mcの選択肢を表示時にシャッフルし、正解indexを再マップ（仕様書§2）
export function shuffleChoices(
  choices: string[],
  answer: number,
  rng: () => number = Math.random,
): { choices: string[]; answer: number } {
  const indices = shuffleArray(choices.map((_, i) => i), rng)
  return {
    choices: indices.map((i) => choices[i]),
    answer: indices.indexOf(answer),
  }
}
```

```ts
// engine/src/lib/mathSegments.ts
export type MathSegment = { kind: 'text' | 'inline' | 'display'; value: string }

// $...$ / $$...$$ を検出して分割する。閉じデリミタが無い場合は平文扱い
export function splitMathSegments(text: string): MathSegment[] {
  const segments: MathSegment[] = []
  let rest = text
  const pattern = /\$\$([\s\S]+?)\$\$|\$([^$\n]+?)\$/
  while (rest.length > 0) {
    const m = rest.match(pattern)
    if (!m || m.index === undefined) {
      segments.push({ kind: 'text', value: rest })
      break
    }
    if (m.index > 0) segments.push({ kind: 'text', value: rest.slice(0, m.index) })
    if (m[1] !== undefined) segments.push({ kind: 'display', value: m[1] })
    else segments.push({ kind: 'inline', value: m[2] })
    rest = rest.slice(m.index + m[0].length)
  }
  return segments.length > 0 ? segments : [{ kind: 'text', value: '' }]
}
```

```ts
// engine/src/lib/resultSummary.ts
import type { PlayResult } from '../types/play'
import type { QuizPack } from '../types/quiz'

export function formatDuration(ms: number): string {
  const totalSec = Math.round(ms / 1000)
  const m = Math.floor(totalSec / 60)
  const s = totalSec % 60
  return `${m}:${String(s).padStart(2, '0')}`
}

export interface CategoryStat {
  category: string
  correct: number
  total: number
}

export function categoryStats(pack: QuizPack, result: PlayResult): CategoryStat[] {
  const byId = new Map(pack.questions.map((q) => [q.id, q]))
  const stats = new Map<string, { correct: number; total: number }>()
  for (const r of result.records) {
    const q = byId.get(r.questionId)
    if (!q) continue
    const s = stats.get(q.category) ?? { correct: 0, total: 0 }
    s.total += 1
    if (r.correct) s.correct += 1
    stats.set(q.category, s)
  }
  return [...stats.entries()].map(([category, s]) => ({ category, ...s }))
}

// 結果コピー（F5）: テキスト要約→LINE共有兼データ収集
export function buildResultText(pack: QuizPack, result: PlayResult): string {
  const total = result.records.length
  const correct = result.records.filter((r) => r.correct).length
  const rate = total === 0 ? 0 : Math.round((correct / total) * 100)
  const cats = categoryStats(pack, result)
    .map((c) => `${c.category} ${c.correct}/${c.total}`)
    .join('・')
  return [
    `【武藤模試】${pack.title}`,
    `正答率 ${rate}%（${correct}/${total}問） 所要時間 ${formatDuration(result.totalMs)}`,
    cats,
  ].join('\n')
}
```

```tsx
// engine/src/App.tsx
import { useState } from 'react'
import { HashRouter, Routes, Route, useNavigate } from 'react-router-dom'
import type { PlayResult } from './types/play'
import type { Question, QuizPack } from './types/quiz'
import { loadSamplePacks } from './data/loadSamplePacks'
import { ToastHost } from './components/Toast'
import { HomePage } from './pages/HomePage'
import { PlayPage } from './pages/PlayPage'
import { ResultPage } from './pages/ResultPage'

const packs = loadSamplePacks()

function AppRoutes() {
  const navigate = useNavigate()
  const [activePack, setActivePack] = useState<QuizPack>(packs[0])
  const [playQuestions, setPlayQuestions] = useState<Question[]>(packs[0].questions)
  const [lastResult, setLastResult] = useState<PlayResult | null>(null)

  const startPlay = (pack: QuizPack, questions: Question[]) => {
    setActivePack(pack)
    setPlayQuestions(questions)
    navigate('/play')
  }

  return (
    <Routes>
      <Route path="/" element={<HomePage packs={packs} onStart={(p) => startPlay(p, p.questions)} />} />
      <Route
        path="/play"
        element={
          <PlayPage
            pack={activePack}
            questions={playQuestions}
            onFinish={(result) => { setLastResult(result); navigate('/result') }}
          />
        }
      />
      <Route
        path="/result"
        element={
          lastResult ? (
            <ResultPage
              pack={activePack}
              result={lastResult}
              onRetryWrong={(wrong) => startPlay(activePack, wrong)}
              onBackHome={() => navigate('/')}
            />
          ) : (
            <HomePage packs={packs} onStart={(p) => startPlay(p, p.questions)} />
          )
        }
      />
    </Routes>
  )
}

export default function App() {
  return (
    <HashRouter>
      <div className="max-w-2xl mx-auto p-4 min-h-screen">
        <AppRoutes />
        <ToastHost />
      </div>
    </HashRouter>
  )
}
```
